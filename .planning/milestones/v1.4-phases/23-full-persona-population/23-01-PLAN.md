def _parse_hexaco(soul_text: str) -> dict[str, float]:
    """Extract hexaco_6 values from SOUL.md text."""
    match = re.search(r"```yaml\s*\n(.*?)```",
                      soul_text.split("## Personality Profile")[1], re.DOTALL)

    return parsed["hexaco_6"]
